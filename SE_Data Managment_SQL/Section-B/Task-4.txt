CREATE DATABASE foodapp_task4;

USE foodapp_task4;

CREATE TABLE customers (
    customer_id INT PRIMARY KEY,
    customer_name VARCHAR(100),
    wallet_balance DECIMAL(10,2)
);

CREATE TABLE orders (
    order_id INT AUTO_INCREMENT PRIMARY KEY,
    customer_id INT,
    restaurant_id INT,
    amount DECIMAL(10,2),
    order_date DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (customer_id)
        REFERENCES customers(customer_id)
);

DELIMITER //

CREATE PROCEDURE place_order(
    IN p_customer_id INT,
    IN p_restaurant_id INT,
    IN p_amount DECIMAL(10,2)
)
BEGIN
    DECLARE current_balance DECIMAL(10,2);

    START TRANSACTION;

    SELECT wallet_balance
    INTO current_balance
    FROM customers
    WHERE customer_id = p_customer_id
    FOR UPDATE;

    IF current_balance < p_amount THEN
        ROLLBACK;
        SELECT 'Insufficient wallet balance. Order cancelled.' AS message;
    ELSE
        UPDATE customers
        SET wallet_balance = wallet_balance - p_amount
        WHERE customer_id = p_customer_id;

        INSERT INTO orders
        (customer_id, restaurant_id, amount)
        VALUES
        (p_customer_id, p_restaurant_id, p_amount);

        COMMIT;
        SELECT 'Order placed successfully.' AS message;
    END IF;
END //

DELIMITER ;
